# =============================================================================
# AdvTG — pinned dependencies for the full 4-stage pipeline
# Target environment: Python 3.11, single NVIDIA GPU (Colab T4 works), CUDA 12.1.
#   - Python 3.13 (current Colab default) CANNOT build this stack — use 3.11.
#   - The paper's testbed was Ubuntu 22.04.1 + Python 3.11 + an RTX 4090.
#
# This is the SAME stack the notebook's "Install" cell uses. Every stage shares it,
# so the whole notebook runs in one kernel with no restarts.
#
# ------------------------------------------------------------------------------
# HOW TO INSTALL — order and flags matter, so do NOT `pip install -r` this blindly.
# Run these three commands in order (this is exactly what the notebook does):
#
#   pip install -r requirements.txt
#   pip install "torch==2.3.1" "torchvision==0.18.1"      # re-assert torch AFTER the above
#   pip install --no-deps "xformers==0.0.26.post1"
#
# Why the last two are separate:
#   * torch 2.3.1 must be (re)installed WITH its deps so it pulls the matching CUDA
#     runtime (cuDNN 8 + triton 2.3.1). Colab ships torch 2.6; this downgrades it.
#     Skipping its deps gives "libcudnn.so.8: cannot open shared object file".
#   * xformers 0.0.26.post1 declares a hard `torch==2.3.0` pin. Installed normally it
#     drags torch back to 2.3.0; --no-deps keeps our 2.3.1 (ABI-compatible within 2.3.x).
# =============================================================================

# NOTHING required. It uses only the Python standard library

# ---- Stage 2 — Detector training (DL/) -------------------------------------
# Token-level (TextCNN / CNN-LSTM / DNN) + image detectors; writes model_configs pickles.
transformers==4.44.2       # BERT tokenizer + TrainingArguments; also the version Stages 3/4 need
datasets==2.21.0           # DL/data_processing.py: `from datasets import Dataset`
scikit-learn>=1.3.0        # DL/metrics.py: accuracy / precision / recall / f1
# torch + numpy are also required here but are pinned in the shared section below.
# (torch is version-agnostic for Stage 2 — it works on any 2.x — but we pin 2.3.1 so the
#  whole notebook shares one stack; numpy comes transitively, resolves to 1.26.x.)


# ---- Stage 3 — LLM finetuning (Unsloth LoRA SFT of Llama-3-8B) --------------
unsloth==2024.8            # the finetuner. Constrains the rest: needs trl<0.9, xformers<0.0.27, torch<=2.3
trl==0.8.6                 # SFTTrainer (Stage 3) AND classic PPOTrainer (Stage 4). <0.9 for unsloth 2024.8
peft>=0.12.0               # LoRA adapters
bitsandbytes==0.43.1       # 4-bit quantization. Newer bnb needs torch>=2.4 (torch.library.register_fake)
accelerate==0.33.0         # newer accelerate's dispatch_model calls model.to() on a 4-bit model -> error
# xformers==0.0.26.post1   # unsloth's attention backend — install LAST with --no-deps (see header)


# ---- Stage 4 — PPO adversarial generation (RL-Adv/) ------------------------

# shared torch/accelerate/bitsandbytes above. The PPO policy (pythia-160m, or the Stage-3
# Llama LoRA) is downloaded at runtime, not pip-installed.
